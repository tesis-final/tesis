! =============================================================================
! ATLAS — atlas-rt01, FASE 9: filtrado entre VLAN (Tabla 44) con ACL extendidas
! Se pega DESPUÉS de atlas-rt01-base.txt (Fase 8), en "configure terminal".
!
! Por qué ACL y no firewall por zonas en este router: es el plan B que ya estaba
! decidido en DEC-04 ("ACL extendidas entrantes por subinterfaz con el mismo
! orden; el retorno TCP con established"). Se usa porque es lo que Packet Tracer
! acepta sin sorpresas. El firewall CON ESTADO está en atlas-fw01 (ZBF), entre
! la red interna e Internet, que es lo que pide el Hito 2 de Conectividad.
!
! Cada ACL se aplica ENTRANDO al router desde su VLAN ("lo que esa VLAN puede
! pedir"). Las líneas "established" dejan pasar solo respuestas de conexiones
! que ya se permitieron del otro lado.
!
! Reglas: R1..R12 = Tabla 44 del PDF. M2 = tabla de políticas del profesor
! (LAN -> servidor 22 denegar). R9b = sede satélite por el túnel IPsec (agregada
! para la VPN opcional del Hito 2, solo HTTPS a ATLAS).
!
! Si Packet Tracer marca error en una línea que termina en "log", pegala de
! nuevo sin la palabra "log".
! =============================================================================
!
! ---------------- VLAN 10 · Administración ----------------------------------
ip access-list extended ACL_VLAN10_ENTRADA
 remark DHCP: pedidos de IP (el router los reenvia al DC)
 permit udp any host 255.255.255.255 eq 67
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 67
 remark R10 MySQL nunca / M2 SSH al servidor desde la LAN, no
 deny tcp any host 172.16.40.10 eq 3306 log
 deny tcp any host 172.16.40.10 eq 22 log
 remark R1 HTTPS a ATLAS
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.10 eq 443
 remark R3 SMB a srv01
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.10 eq 445
 remark R2 servicios de dominio al DC
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 53
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 53
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 88
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 88
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 464
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 464
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 389
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 389
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 636
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 range 3268 3269
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 445
 permit udp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 123
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 eq 135
 permit tcp 172.16.10.0 0.0.0.255 host 172.16.40.20 range 49152 65535
 remark R6 respuestas a la estacion de gestion (SSH y RDP)
 permit tcp 172.16.10.0 0.0.0.255 eq 22 172.16.50.0 0.0.0.255 established
 permit tcp 172.16.10.0 0.0.0.255 eq 3389 172.16.50.0 0.0.0.255 established
 remark R12 todo lo demas hacia adentro, negado y registrado
 deny ip any 172.16.0.0 0.0.255.255 log
 deny ip any 10.8.0.0 0.0.0.255 log
 deny ip any 192.168.100.0 0.0.0.255 log
 remark R5 navegar en Internet
 permit tcp 172.16.10.0 0.0.0.255 any eq 80
 permit tcp 172.16.10.0 0.0.0.255 any eq 443
 remark R12
 deny ip any any log
!

! ---------------- Desde atlas-fw01 (Internet y túneles) ---------------------
ip access-list extended ACL_DESDE_FW01

 deny tcp any host 172.16.40.10 eq 3306 log
 remark Respuestas de Internet a lo que se pidio de adentro (R5, R7)

 remark R9 VPN docentes hacia ATLAS

 remark R9b sede satelite (tunel IPsec) hacia ATLAS
 permit tcp 192.168.100.0 0.0.0.255 host 172.16.40.10 eq 443
 remark Monitoreo: registros (syslog) de atlas-fw01 hacia srv01

 remark R12
